\section{Introdução}

\begin{frame}{Objetivo do minicurso}{Introdução}
\begin{itemize}\setlength{\itemsep}{.7em}
\item Formular problemas de redes a partir dos dados disponíveis.
\item Localização: sinais de vários gateways \(\rightarrow\) posição.
\item Forecasting: histórico de um enlace \(\rightarrow\) RSSI da próxima hora.
\end{itemize}
\end{frame}

\begin{frame}{Da pergunta à avaliação}{Introdução}
\begin{itemize}\setlength{\itemsep}{.7em}
\item Definir observação, entradas e resposta.
\item Separar os dados conforme o uso esperado.
\item Comparar com uma referência simples; interpretar o erro na aplicação.
\end{itemize}
\end{frame}

\begin{frame}{Informação disponível}{Introdução}
\begin{itemize}\setlength{\itemsep}{.7em}
\item Após a recepção: RSSI e SNR podem ajudar a localizar o transmissor.
\item Antes da transmissão: essas medidas ainda não existem.
\item GPS fornece o rótulo, não uma entrada para localizar.
\end{itemize}
\end{frame}

\section{Localização}

\begin{frame}{Localização em Antwerp}{Notebook 01 \textperiodcentered{} 40 min}
\begin{itemize}\setlength{\itemsep}{.7em}
\item 12.000 pacotes; RSSI de 39 gateways com coordenadas conhecidas.
\item Dias anteriores: treino; intermediários: validação; posteriores: teste.
\item Explorar a coleta antes dos modelos.
\end{itemize}
\end{frame}

\section{Forecasting}

\begin{frame}{Previsão de RSSI}{Notebook 02 \textperiodcentered{} 40 min}
\begin{itemize}\setlength{\itemsep}{.7em}
\item Após fechar a hora t, estimar o RSSI médio de t+1.
\item Um enlace, 1.059 horas contínuas.
\item Ler a série antes de construir as entradas.
\end{itemize}
\end{frame}

\begin{frame}{Do histórico à tabela}{Notebook 02 \textperiodcentered{} lags}
\begin{itemize}\setlength{\itemsep}{.7em}
\item Entradas: RSSI de 1, 2, 3, 6 e 24 horas atrás.
\item Treino no passado; teste no futuro.
\item Persistência repete o último valor; Ridge combina os lags.
\end{itemize}
\end{frame}

\begin{frame}{Avaliação temporal}{Notebook 02 \textperiodcentered{} resultados}
\centering\includegraphics[width=.92\textwidth,height=.54\textheight,keepaspectratio]{figures/02_forecasting_06.png}
{\par\scriptsize Previsões de uma hora à frente. Dados: Goldoni et al.\par}
\begin{itemize}\footnotesize
\item MAE e RMSE em dB; valores de RSSI em dBm.
\item Uma hora à frente por vez, com o modelo fixo.
\item Observar os erros nos eventos, não apenas a média.
\end{itemize}
\end{frame}

\section{Avaliação}

\begin{frame}{Erros de metodologia}{Discussão \textperiodcentered{} 20 min}
\begin{itemize}\setlength{\itemsep}{.7em}
\item Normalizar com treino e teste contamina a avaliação.
\item Trocar parâmetros olhando o teste transforma o teste em validação.
\item Pacotes próximos não equivalem a situações independentes.
\end{itemize}
\end{frame}

\begin{frame}{Métrica e utilidade}{Discussão}
\begin{itemize}\setlength{\itemsep}{.7em}
\item Erro pequeno em locais frequentes pode esconder falhas em outras regiões.
\item RSSI das recepções não mede taxa de entrega de pacotes.
\item Antes de usar na rede: considerar tolerância, cobertura e custo dos erros.
\end{itemize}
\end{frame}

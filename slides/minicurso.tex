\begin{frame}{Duas frentes, um mesmo método}{O que vamos fazer}
\begin{columns}[T]

$\downarrow$\\

\end{block}

\column{.48\textwidth}
\begin{block}{2. Previsão temporal}
Temos o histórico de um enlace.\\[2mm]
\textbf{Qual será o RSSI na próxima hora?}\\[2mm]
Sinais de vários instantes\\
$\downarrow$\\
valor futuro estimado
\end{block}

\end{columns}

\begin{frame}{Por que essas perguntas interessam a redes?}{Da aplicação ao dado}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Localização de um ativo}
Um equipamento transmite pela rede. Queremos uma indicação de sua posição sem fornecer seu GPS ao estimador.
\end{block}
Serve para indicar uma região?\\
É preciso encontrar uma sala?
\column{.48\textwidth}
\begin{block}{Monitoramento de um enlace}
Um sensor envia telemetria regularmente. Queremos antecipar a evolução de uma medida do sinal.
\end{block}
Há informação no histórico?\\
Ela acrescenta algo ao último valor?
\end{columns}
\vspace{3mm}
\textbf{Limite:} prever RSSI não equivale a prever falha, perda de pacotes ou disponibilidade.
\end{frame}

\section{Forecasting}
\begin{frame}{Agora, o que ainda não chegou?}{Notebook 02 — seções 0 a 2}
\coursefigure{02_forecasting_01.png}{Histórico do enlace. Dados: Goldoni et al.; exploração restrita ao histórico de treino.}
\textbf{Situação:} após fechar uma hora de observações, estimar o RSSI médio da próxima.\\
Se a série varia lentamente, repetir o último valor já pode funcionar.
\end{frame}

\begin{frame}{Uma série também vira uma tabela supervisionada}{Notebook 02 — seção 3; exemplo fictício}
\centering
\begin{tabular}{c r r r}
\toprule
Hora-alvo & RSSI anterior & RSSI de duas horas atrás & Resposta\\
\midrule
10 h & $-82$ & $-80$ & $-81$\\
11 h & $-81$ & $-82$ & ainda desconhecida\\
\bottomrule
\end{tabular}
\vspace{4mm}
\[
X(u)=[y(u-1),y(u-2),y(u-3),y(u-6),y(u-24)]
\qquad\longrightarrow\qquad y(u)
\]
\begin{itemize}
\item Cada \emph{lag} é uma observação passada.
\item O target é um número futuro: regressão com dependência temporal.
\item Não fornecer o RSSI da hora-alvo como entrada.
\end{itemize}
\end{frame}

\begin{frame}{Respeitar a ordem dos acontecimentos}{Notebook 02 — seção 4}
\centering
\begin{tikzpicture}
\node[box,text width=4cm] (tr) at (0,0) {828 exemplos de treino\\aprender modelo e escala};
\node[box,text width=4cm] (te) at (6,0) {207 exemplos de teste\\aplicar sem reajustar};
\draw[arrow] (tr)--node[above]{tempo}(te);
\end{tikzpicture}
\vspace{4mm}
\begin{itemize}
\item 1.059 horas contínuas; 24 horas iniciais formam o histórico dos lags.
\item Formar lags com passado é diferente de ajustar um scaler com todos os dados.
\item Testar no futuro simula a direção em que a previsão será usada.
\end{itemize}
\end{frame}

\begin{frame}{O que o modelo precisa superar?}{Notebook 02 — seção 5}
\begin{columns}[T]
\column{.46\textwidth}
\begin{block}{Persistência}
\[
\widehat y(u)=y(u-1)
\]
Repete o último valor disponível.\\[2mm]
Sem treinamento; uma referência natural para uma série pouco variável.
\end{block}
\column{.50\textwidth}
\begin{block}{Regressão Ridge}
\[
\widehat y(u)=b+\sum_j w_j\,y(u-k_j)
\]
Aprende uma combinação dos lags, com penalização dos coeficientes.\\[2mm]
Pesos e escala vêm apenas do treino.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Uma hora à frente, repetidamente}{O teste não é uma previsão única de 207 horas}
\centering
\begin{tikzpicture}
\node[box,text width=2.4cm] (a) at (0,0) {Fechar hora $t$\\observar $y(t)$};
\node[box,text width=2.4cm] (b) at (3.5,0) {Prever\\$y(t+1)$};
\node[box,text width=2.4cm] (c) at (7,0) {Fechar $t+1$\\receber valor real};
\node[box,text width=2.4cm] (d) at (10.5,0) {Prever\\$y(t+2)$};
\draw[arrow] (a)--(b); \draw[arrow] (b)--(c); \draw[arrow] (c)--(d);
\end{tikzpicture}
\vspace{5mm}
\begin{itemize}
\item Uma observação de teste pode virar histórico \textbf{depois} que chega.
\item O modelo permanece fixo; não fazemos novo treinamento a cada hora.
\item A média de uma hora só está disponível após o fechamento da janela.
\end{itemize}
\end{frame}

\begin{frame}{A previsão melhora a referência?}{Notebook 02 — seção 6}
\coursefigure{02_forecasting_06.png}{Teste sequencial de uma hora à frente. Dados: Goldoni et al.}
\begin{itemize}
\item RSSI em dBm; MAE e RMSE em dB.
\item Real $-80$ dBm, previsão $-83$ dBm: erro absoluto de 3 dB.
\item Olhar os picos de erro, não só a média.
\end{itemize}
\end{frame}

\small
\begin{tabular}{p{2.8cm}p{4.8cm}p{4.4cm}}

\section{Discussão}
\begin{frame}{Em qual caso há vazamento?}{20 minutos — discutir o instante e a separação}
\begin{enumerate}
\item Localizar após receber, usando SNR daquele pacote.
\item Prever o RSSI de amanhã usando a temperatura \emph{observada} amanhã.
\item Ajustar o scaler com treino e teste juntos.
\item Usar o RSSI recebido às 10 h para prever às 11 h.
\end{enumerate}
\vspace{4mm}
Antes de responder: \textbf{essa informação estaria disponível quando a decisão fosse tomada?}
\end{frame}

\begin{frame}{As respostas dependem do contrato}{Retomar os quatro casos}
\begin{itemize}
\item \textbf{1 e 4:} informações já disponíveis, conforme o instante definido.
\item \textbf{2:} usa futuro observado; uma previsão meteorológica emitida antes seria outra informação.
\item \textbf{3:} o teste participa do ajuste da transformação.
\end{itemize}
\begin{block}{Outro cuidado}
Escolher filtros, parâmetros ou a semente repetidamente pelo resultado de teste
faz o teste participar do desenvolvimento.
\end{block}
\end{frame}

\begin{frame}{Uma boa métrica não encerra a análise}{O que faltaria para uma aplicação real?}
\begin{itemize}
\item \textbf{Representatividade:} quais lugares, dispositivos e períodos ficaram de fora?
\item \textbf{Generalização:} treino bom e teste ruim podem indicar overfitting ou mudança de condições.
\item \textbf{Utilidade:} qual erro é tolerável? Quanto custa uma decisão errada?
\item \textbf{Observabilidade:} só vemos recepções; não inferimos PDR sem as transmissões perdidas.
\end{itemize}
\end{frame}
